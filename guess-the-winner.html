},

// 🔧 APP LOGIC — no need to edit below

const guesses = {};
let current = 0; // 0-indexed
function renderCurrent() {
const exp = EXPERIMENTS[current];
const num = current + 1;
const total = EXPERIMENTS.length;
document.getElementById("cardLabel").textContent = `Experiment ${num} of ${total}`;
document.getElementById("cardTitle").textContent = exp.title;
document.getElementById("cardSubtitle").textContent = exp.subtitle || "";
document.getElementById("labelA").textContent = exp.labelA;
document.getElementById("labelB").textContent = exp.labelB;
// Restore previous pick if user somehow navigates back (defensive)
const prev = guesses[num];
document.getElementById("btnA").classList.toggle("selected", prev === "A");
document.getElementById("btnB").classList.toggle("selected", prev === "B");
const continueBtn = document.getElementById("continueBtn");
continueBtn.textContent = num === total ? "Submit guesses →" : "Continue →";
continueBtn.classList.toggle("visible", !!prev);
updateProgress();
}
function pick(choice) {
const num = current + 1;
guesses[num] = choice;
document.getElementById("btnA").classList.toggle("selected", choice === "A");
document.getElementById("btnB").classList.toggle("selected", choice === "B");
document.getElementById("continueBtn").classList.add("visible");
updateProgress();
}
function advance() {
const card = document.getElementById("expCard");
// Slide out current card
card.classList.add("leaving");
setTimeout(() => {
card.classList.remove("leaving");
current++;
if (current >= EXPERIMENTS.length) {
// Show submit screen
card.style.display = "none";
document.getElementById("submit-screen").style.display = "block";
updateProgress();
} else {
renderCurrent();
}
}, 250);
}
function updateProgress() {
const answered = Object.keys(guesses).length;
const total = EXPERIMENTS.length;
const pct = (answered / total) * 100;
document.getElementById("progressBar").style.width = pct + "%";
document.getElementById("progressLabel").textContent = `${answered} / ${total}`;
}
function calcScore() {
let score = 0;
EXPERIMENTS.forEach((exp, i) => {
if (guesses[i + 1] === exp.winner) score++;
});
return score;
}
async function handleSubmit() {
const name = document.getElementById("nameInput").value.trim();
const email = document.getElementById("emailInput").value.trim();
const warn = document.getElementById("warningMsg");
if (!name || !email) {
warn.textContent = "Please enter your name and email.";
warn.classList.add("visible");
return;
}
if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) {
warn.textContent = "Please enter a valid email address.";
warn.classList.add("visible");
return;
}
warn.classList.remove("visible");
const btn = document.getElementById("submitBtn");
btn.disabled = true;
btn.textContent = "Submitting…";
const score = calcScore();
const answered = Object.keys(guesses).length;
const payload = { name, email, score, total: EXPERIMENTS.length, answered };
EXPERIMENTS.forEach((_, i) => { payload[`exp${i + 1}`] = guesses[i + 1] || ""; });
try {
if (GOOGLE_SCRIPT_URL && GOOGLE_SCRIPT_URL !== "YOUR_APPS_SCRIPT_URL_HERE") {
await fetch(GOOGLE_SCRIPT_URL, {
method: "POST",
mode: "no-cors",
headers: { "Content-Type": "application/json" },
body: JSON.stringify(payload),
});
} else {
console.log("Submission (no script URL set):", payload);
}
showThankYou(score);
} catch (err) {
btn.disabled = false;
btn.textContent = "Submit my guesses";
warn.textContent = "Something went wrong. Please try again.";
warn.classList.add("visible");
}
}
function showThankYou(score) {
document.getElementById("submit-screen").style.display = "none";
const ty = document.getElementById("thank-you");
ty.style.display = "block";
document.getElementById("scoreDisplay").textContent =
`Your score: ${score} / ${EXPERIMENTS.length}`;
}
// Init
renderCurrent();
